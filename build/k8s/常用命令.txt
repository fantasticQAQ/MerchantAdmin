### 导出镜像
docker save datalust/seq:latest -o F:\Docker\seq.tar
传输镜像
scp F:\Docker\sqlserver.tar ubuntu@你的IP:/home/ubuntu/
###  导入镜像
docker load < sqlserver.tar

# 临时用社区代理拉（不改 daemon）
docker pull mcr.dockerproxy.com/mssql/server:2019-latest
# 拉完改 tag
docker tag mcr.dockerproxy.com/mssql/server:2019-latest mcr.microsoft.com/msssql/server:2019-latest

代理转发：
kubectl port-forward -n merchant svc/sqlserver 1433:1433
kubectl port-forward -n merchant svc/merchantadmin 80:80
kubectl port-forward -n merchant svc/merchant-api 5002:5002
kubectl port-forward -n merchant svc/seq 5341:5341

停止重启
kubectl scale deployment merchant-api -n merchant --replicas=0
查看service的Ip
kubectl get svc -n merchant

重新应用yaml文件
kubectl apply -f 05-secrets.yaml
kubectl apply -f 07-merchant-api.yaml
重启pod
kubectl rollout restart deployment/merchant-api -n merchant

查看pods状态
kubectl get pods -n merchant
查看pods的日志
kubectl logs merchant-api-548968fc87-82cj7 -n merchant
让 K8s 用新镜像重建 Pod
kubectl rollout restart deployment merchant-api identity-api -n merchant
# 只删 Pod（Deployment 会立刻重建一个新的）
kubectl delete pod -l app=merchant-api -n merchant
# 删 Deployment + 它管的所有 Pod（彻底删掉）
kubectl delete deployment merchant-api -n merchant
# 删 Service（如果有的话）
kubectl delete svc merchant-api -n merchant

# 1. 重新构建前端镜像（不用缓存）

# 2. 启动前端容器
docker-compose up -d frontend
下载ingress-nginx
kubectl apply -f https://raw.githubusercontent.com/kubernetes/ingress-nginx/main/deploy/static/provider/cloud/deploy.yaml

#1、彻底删除数据库（停全部服务 + 删数据卷：sqlserver 数据、Redis/Seq 数据一并清空）
docker compose down -v

#2、构建所有镜像
docker compose build

#3、只起基础设施，等 sqlserver 健康
docker compose up -d sqlserver redis rabbitmq seq

#4、迁移
docker compose --profile migrate build db-migrator
docker compose --profile migrate run --rm merchant-migrator
docker compose --profile migrate run --rm identity-migrator

#5、起全部服务
docker compose up -d


# 看迁移日志
docker compose --profile migrate logs merchant-migrator
docker compose --profile migrate logs identity-migrator
# 看某个服务日志
docker compose logs -f merchant-api
